--
-- This extension allows saving data to specific object for it's lifetime in current runtime session
-- Supports Map, Player, Creature, GameObject
--
-- SetData sets a value
--  obj:SetData(key, val)
--
-- GetData gets the data table or a specific value by key from it
--  local tbl = obj:GetData()
--  local val = obj:GetData(key)
--

local pairs = pairs

-- ALE TypeId enum values (from Object:GetTypeId)
local TYPEID_UNIT       = 3  -- Creature
local TYPEID_PLAYER     = 4
local TYPEID_GAMEOBJECT = 5

-- Map TypeId to string for variable store keys
-- Note: Map objects don't have GetTypeId, handled separately
local typeIdToName = {
    [TYPEID_UNIT]       = "Creature",
    [TYPEID_PLAYER]     = "Player",
    [TYPEID_GAMEOBJECT] = "GameObject",
}

local variableStores = {
    Map = {},
    Player = {},
    Creature = {},
    GameObject = {},
}

-- {{{ DestroyMapData
local function DestroyMapData(event, obj)
    local map = obj:GetMapId()
    local inst = obj:GetInstanceId()
    for k,v in pairs(variableStores) do
        local mapdata = v[map]
        if mapdata then
            mapdata[inst] = nil
        end
    end
end
-- }}}

-- {{{ DestroyObjData
local function DestroyObjData(event, obj)
    -- ALE uses GetTypeId() instead of GetObjectType()
    local typeId = obj:GetTypeId()
    local otype = typeIdToName[typeId]
    if not otype then return end  -- unknown type

    local guid = obj:GetGUIDLow()

    if otype == "Player" then
        variableStores[otype][guid] = nil
        return
    end

    local map = obj:GetMapId()
    local inst = obj:GetInstanceId()
    local mapdata = variableStores[otype][map]
    if mapdata then
        local instancedata = mapdata[inst]
        if instancedata then
            instancedata[guid] = nil
        end
    end
end
-- }}}

-- {{{ GetData
local function GetData(self, field)
    -- ALE uses GetTypeId() instead of GetObjectType()
    local typeId = self:GetTypeId()
    local otype = typeIdToName[typeId]
    if not otype then return nil end  -- unknown type

    local guid = self:GetGUIDLow()
    local varStore = variableStores[otype]

    if otype == "Player" then
        varStore[guid] = varStore[guid] or {}
        if field ~= nil then
            return varStore[guid][field]
        end
        return varStore[guid]
    end

    local map = self:GetMapId()
    local inst = self:GetInstanceId()
    varStore[map] = varStore[map] or {}
    varStore[map][inst] = varStore[map][inst] or {}
    varStore[map][inst][guid] = varStore[map][inst][guid] or {}

    if field ~= nil then
        return varStore[map][inst][guid][field]
    end
    return varStore[map][inst][guid]
end
-- }}}

-- {{{ SetData
local function SetData(self, field, val)
    -- ALE uses GetTypeId() instead of GetObjectType()
    local typeId = self:GetTypeId()
    local otype = typeIdToName[typeId]
    if not otype then return end  -- unknown type

    local guid = self:GetGUIDLow()
    local varStore = variableStores[otype]

    if otype == "Player" then
        varStore[guid] = varStore[guid] or {}
        varStore[guid][field] = val
        return
    end

    local map = self:GetMapId()
    local inst = self:GetInstanceId()
    varStore[map] = varStore[map] or {}
    varStore[map][inst] = varStore[map][inst] or {}
    varStore[map][inst][guid] = varStore[map][inst][guid] or {}

    varStore[map][inst][guid][field] = val
end
-- }}}

-- {{{ Assign methods to globals
for k,v in pairs(variableStores) do
    _G[k].GetData = GetData
    _G[k].SetData = SetData
end
-- }}}

-- {{{ Event registrations for automatic cleanup
-- MOVED TO object-variables-events.lua (Issue 332)
-- .ext files load before hooks are ready, causing registry corruption
-- .lua files load after hooks are initialized
-- }}}

-- {{{ ObjectVariables namespace (for compatibility with code that calls cleanup manually)
ObjectVariables = {}

-- Expose destroy functions for event registration in .lua files
ObjectVariables._destroyObjData = DestroyObjData
ObjectVariables._destroyMapData = DestroyMapData

function ObjectVariables.cleanupPlayer(player)
    if not player then return end
    local guid = player:GetGUIDLow()
    if guid then
        variableStores.Player[guid] = nil
    end
end

function ObjectVariables.cleanupCreature(creature)
    if not creature then return end
    local guid = creature:GetGUIDLow()
    local mapId = creature:GetMapId()
    local instId = creature:GetInstanceId()

    local mapdata = variableStores.Creature[mapId]
    if mapdata then
        local instancedata = mapdata[instId]
        if instancedata then
            instancedata[guid] = nil
        end
    end
end

function ObjectVariables.cleanupGameObject(gameobject)
    if not gameobject then return end
    local guid = gameobject:GetGUIDLow()
    local mapId = gameobject:GetMapId()
    local instId = gameobject:GetInstanceId()

    local mapdata = variableStores.GameObject[mapId]
    if mapdata then
        local instancedata = mapdata[instId]
        if instancedata then
            instancedata[guid] = nil
        end
    end
end

function ObjectVariables.cleanupGameObjectByLocation(mapId, instanceId, guid)
    if not mapId or not instanceId or not guid then return end

    local mapdata = variableStores.GameObject[mapId]
    if mapdata then
        local instancedata = mapdata[instanceId]
        if instancedata then
            instancedata[guid] = nil
        end
    end
end

function ObjectVariables.cleanupMap(mapId, instanceId)
    for k,v in pairs(variableStores) do
        local mapdata = v[mapId]
        if mapdata then
            mapdata[instanceId] = nil
        end
    end
end

function ObjectVariables.getStats()
    local stats = {
        players = 0,
        creatures = 0,
        gameobjects = 0,
    }
    for guid, _ in pairs(variableStores.Player) do
        stats.players = stats.players + 1
    end
    for mapId, mapdata in pairs(variableStores.Creature) do
        for instId, instdata in pairs(mapdata) do
            for guid, _ in pairs(instdata) do
                stats.creatures = stats.creatures + 1
            end
        end
    end
    for mapId, mapdata in pairs(variableStores.GameObject) do
        for instId, instdata in pairs(mapdata) do
            for guid, _ in pairs(instdata) do
                stats.gameobjects = stats.gameobjects + 1
            end
        end
    end
    return stats
end
-- }}}

print("[ObjectVariables] Loaded with automatic cleanup via events")
