```lua
-- Assuming we have the following objects (replace with your data)

    { x = 0.5, y = 0.3 },

    -- ... more objects here

-- Create a grid with dimensions 10x10 (you can change this as needed)
local gridSizeX = 10
local gridSizeY = 10
local grid = {}
for i=1,gridSizeX do
    grid[i] = {}
end

-- Sort objects into the grid based on their coordinates
for _,object in ipairs(objects) do
    -- Calculate which grid cell this object belongs to (scaling the coordinates to fit the grid size)
    local xIndex = math.floor(object.x * gridSizeX) + 1
    local yIndex = math.floor(object.y * gridSizeY) + 1

    -- Add the object to that cell's array (creating it if necessary)
    if not grid[xIndex][yIndex] then
        grid[xIndex][yIndex] = {}
    end
    table.insert(grid[xIndex][yIndex], object)
end
```

Now, `grid` is a 2D array where each cell contains an array of objects that fall within its boundaries. You can iterate over the grid and process the objects accordingly.
